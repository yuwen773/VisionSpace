function WechatCapsule({ dark }) {
  const fg = dark ? '#000' : '#fff';
  const bg = dark ? 'rgba(255,255,255,0.6)' : 'rgba(0,0,0,0.15)';
  const bd = dark ? 'rgba(0,0,0,0.08)' : 'rgba(255,255,255,0.25)';
  return (
    <div data-mp-capsule="" data-lint-skip="" style={{ ...wechatFrameStyles.capsule, background: bg, border: `0.5px solid ${bd}` }}>
      <div style={wechatFrameStyles.capsuleHalf}>
        <svg width="20" height="6" viewBox="0 0 20 6"><circle cx="3" cy="3" r="2" fill={fg} /><circle cx="10" cy="3" r="2.6" fill={fg} /><circle cx="17" cy="3" r="2" fill={fg} /></svg>
      </div>
      <div style={{ ...wechatFrameStyles.capsuleSep, background: bd }} />
      <div style={wechatFrameStyles.capsuleHalf}>
        <svg width="18" height="18" viewBox="0 0 18 18"><circle cx="9" cy="9" r="7" fill="none" stroke={fg} strokeWidth="1.6" /><circle cx="9" cy="9" r="2.6" fill={fg} /></svg>
      </div>
    </div>
  );
}

function WechatTabBar({ list, active, onChange, color, selectedColor, backgroundColor, borderColor }) {
  return (
    <div data-mp-tabbar="native" data-lint-skip="" style={{ ...wechatFrameStyles.tabBar, background: backgroundColor || '#fff', borderTop: `0.5px solid ${borderColor || 'rgba(0,0,0,0.12)'}` }}>
      {list.map(it => {
        const on = it.text === active;
        return (
          <div key={it.text} style={{ ...wechatFrameStyles.tabItem, color: on ? selectedColor : color }} onClick={() => onChange && onChange(it.text)}>
            {it.icon ? it.icon(on) : <div style={{ width: 5, height: 5, borderRadius: 999, background: 'currentColor', opacity: on ? 1 : 0.45 }} />}
            <span style={{ letterSpacing: '0.08em' }}>{it.text}</span>
          </div>
        );
      })}
    </div>
  );
}

function WechatFrame({ children, page = 'pages/index/index', title = '', nav = 'default', navColor = '#ffffff', navText = 'black',
  back = false, onBack, tab, tabBar, onTabChange, footer, background = '#ffffff', width = 375, height = 812, time = '9:41', tabColors }) {
  const statusColor = navText === 'white' ? '#fff' : '#000';
  const hasNativeTab = tabBar && tabBar.list && tabBar.list.length;
  const top = 84;
  const bottom = hasNativeTab ? 83 : 0;
  const footRef = React.useRef(null);
  const [footH, setFootH] = React.useState(0);
  React.useLayoutEffect(() => { const el = footRef.current; setFootH(el ? el.offsetHeight : 0); });
  return (
    <div style={wechatFrameStyles.wrapper}>
      <div style={{ ...wechatFrameStyles.screen, width, height, background }} data-mp-page={page} data-mp-title={title} data-mp-nav={nav} data-mp-tab={tab || ''}>
        <div data-mp-chrome="" data-lint-skip="" style={{ ...wechatFrameStyles.statusBar, color: statusColor }}>
          <span>{time}</span>
          <svg width="66" height="12" viewBox="0 0 66 12" fill={statusColor}>
            <rect x="0" y="8" width="3" height="4" rx="1" /><rect x="5" y="6" width="3" height="6" rx="1" /><rect x="10" y="3" width="3" height="9" rx="1" /><rect x="15" y="0" width="3" height="12" rx="1" />
            <path d="M30 11.2a1.2 1.2 0 100-2.4 1.2 1.2 0 000 2.4zM25.6 6.4a6.4 6.4 0 018.8 0l-1.1 1.1a4.8 4.8 0 00-6.6 0zM23 3.8a10 10 0 0114 0l-1.1 1.1a8.4 8.4 0 00-11.8 0z" />
            <rect x="41" y="0.75" width="22" height="10.5" rx="3" fill="none" stroke={statusColor} strokeOpacity="0.45" strokeWidth="1" /><rect x="43" y="2.75" width="16" height="6.5" rx="1.5" /><rect x="64" y="4" width="1.5" height="4" rx="0.75" fillOpacity="0.45" />
          </svg>
        </div>
        {nav === 'default' && <div style={{ position: 'absolute', top: 0, left: 0, right: 0, height: top, background: navColor, zIndex: 25 }} />}
        {nav === 'default' && (
          <div style={{ ...wechatFrameStyles.nav, color: statusColor }}>
            {back && (
              <div style={wechatFrameStyles.back} onClick={onBack}>
                <svg width="12" height="20" viewBox="0 0 12 20"><path d="M10 2L2 10l8 8" fill="none" stroke={statusColor} strokeWidth="1.6" strokeLinecap="round" strokeLinejoin="round" /></svg>
              </div>
            )}
            <div style={wechatFrameStyles.navTitle}>{title}</div>
          </div>
        )}
        <WechatCapsule dark={navText !== 'white'} />
        <div data-mp-scroll="" style={{ position: 'absolute', top: nav === 'default' ? top : 0, left: 0, right: 0, bottom: bottom + footH, overflow: 'auto' }}>
          {children}
        </div>
        {footer && <div ref={footRef} data-mp-footer="" style={{ position: 'absolute', left: 0, right: 0, bottom, zIndex: 40, paddingBottom: hasNativeTab ? 0 : 34 }}>{footer}</div>}
        {hasNativeTab && <WechatTabBar {...tabBar} active={tab} onChange={onTabChange} />}
        <div style={{ ...wechatFrameStyles.homeIndicator, background: (!hasNativeTab && navText === 'white') ? 'rgba(255,255,255,0.6)' : 'rgba(0,0,0,0.85)' }} />
      </div>
    </div>
  );
}

/* ---------- 展示板（取自 skill 的 assets/showboard.jsx，layout=hero） ---------- */
function ShowboardLabel({ s, theme, width }) {
  return (
    <div style={{ width, marginBottom: 10 }}>
      <div style={{ fontSize: 15, fontWeight: 700, color: theme.fg, lineHeight: 1.3 }}>{s.label}</div>
      {s.note && <div style={{ fontSize: 14, color: theme.muted, lineHeight: 1.5, marginTop: 2 }}>{s.note}</div>}
    </div>
  );
}
function ShowboardPhone({ s, theme, x, y, scale, phoneW, phoneH }) {
  return (
    <div style={{ position: 'absolute', left: x, top: y, width: phoneW * scale }}>
      <ShowboardLabel s={s} theme={theme} width={phoneW * scale} />
      <div style={{ width: phoneW * scale, height: phoneH * scale }}>
        <div style={{ transform: `scale(${scale})`, transformOrigin: '0 0', width: phoneW }}>{s.node}</div>
      </div>
    </div>
  );
}
function Showboard({ layout = 'hero', theme = {}, title, kicker, lede, screens = [], aside, phoneW = 395, phoneH = 832, footnote }) {
  const t = { bg: '#F2F1EC', fg: '#1B1B1B', muted: '#6B6B66', accent: '#C2410C', font: 'inherit', ...theme };
  const W = 1440, M = 56;
  const mainIdx = Math.max(0, screens.findIndex((s) => s.main));
  const main = screens[mainIdx], rest = screens.filter((_, i) => i !== mainIdx);
  const LABEL = 56;
  const placed = [];
  let asidePos = null, height = 0;
  if (layout === 'hero') {
    const big = 0.95, small = 0.76, my = 250;
    placed.push({ s: main, x: M, y: my, scale: big });
    const colX = [M + phoneW * big + 56, M + phoneW * big + 56 + phoneW * small + 36];
    rest.forEach((s, i) => {
      const col = i % 2, row = Math.floor(i / 2);
      placed.push({ s, x: colX[col], y: my - 40 + (col ? 140 : 0) + row * (phoneH * small + LABEL + 36), scale: small });
    });
    asidePos = { x: colX[1] + phoneW * small + 56, y: my + 40, w: W - M - (colX[1] + phoneW * small + 56) };
  }
  placed.forEach((p) => { height = Math.max(height, p.y + LABEL + phoneH * p.scale + M); });
  return (
    <div data-showboard={layout} style={{ position: 'relative', width: W, height, background: t.bg, color: t.fg, fontFamily: t.font, overflow: 'hidden' }}>
      <div style={{ position: 'absolute', left: M, top: 56, width: 640 }}>
        {kicker && <div style={{ fontSize: 14, color: t.accent, fontWeight: 700, marginBottom: 10 }}>{kicker}</div>}
        <div style={{ fontSize: 56, fontWeight: 800, lineHeight: 1.05 }}>{title}</div>
        {lede && <div style={{ fontSize: 16, lineHeight: 1.65, color: t.fg, opacity: 0.86, marginTop: 18 }}>{lede}</div>}
      </div>
      {placed.map((p, i) => <ShowboardPhone key={i} {...p} theme={t} phoneW={phoneW} phoneH={phoneH} />)}
      {aside && asidePos && <div style={{ position: 'absolute', left: asidePos.x, top: asidePos.y, width: asidePos.w }}>{aside}</div>}
      {footnote && <div style={{ position: 'absolute', left: M, bottom: 28, fontSize: 12, color: t.muted }}>{footnote}</div>}
    </div>
  );
}
